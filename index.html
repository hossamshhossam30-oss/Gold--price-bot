async function updatePrices() { try { // 1. جلب سعر الدولار مقابل الجنيه const response = await fetch('https://open.er-api.com/v6/latest/USD'); const data = await response.json(); const usdToEgp = data.rates.EGP; // 2. حساب عيار 24 (سعر الأونصة افتراضي 2400 دولار كمثال) const goldOunceUSD = 2400; const price24 = Math.round((goldOunceUSD / 31.1034768) * usdToEgp); // 3. تحديث الأرقام في الواجهة (DOM) document.getElementById('buy24').textContent = price24.toLocaleString('ar-EG'); document.getElementById('sell24').textContent = Math.round(price24 * 1.02).toLocaleString('ar-EG'); document.getElementById('last-update').textContent = new Date().toLocaleTimeString('ar-EG'); } catch (error) { console.error('خطأ في البيانات:', error); } } // تشغيل التحديث التلقائي كل 30 ثانية setInterval(updatePrices, 30000); // تشغيل الدالة فور تحميل الصفحة أول مرة updatePrices();
