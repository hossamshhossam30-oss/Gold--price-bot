async function updatePrices() { try { // جلب سعر الدولار مقابل الجنيه لحظياً const response = await fetch('https://open.er-api.com/v6/latest/USD'); const data = await response.json(); const usdToEgp = data.rates.EGP; // سعر أونصة الذهب عالمياً (يمكن جلبها أو تثبيت متوسط متحرك) // أو جلب سعر الذهب المباشر عبر API مخصص مثل MetalpriceAPI const goldOunceUSD = 2400; // يتم تعديله أو جلبة ديناميكياً // حساب عيار 24 وباقي العيارات const price24 = Math.round((goldOunceUSD / 31.1034768) * usdToEgp); const price21 = Math.round(price24 * (21 / 24)); const price18 = Math.round(price24 * (18 / 24)); // تحديث الأرقام في عناصر الواجهة (DOM) document.getElementById('buy24').textContent = price24.toLocaleString('ar-EG'); document.getElementById('sell24').textContent = Math.round(price24 * 1.02).toLocaleString('ar-EG'); // تحديث وقت التحديث document.getElementById('last-update').textContent = new Date().toLocaleTimeString('ar-EG'); } catch (error) { console.error('خطأ في جلب البيانات:', error); } } // تشغيل التحديث كل 30 ثانية تلقائياً setInterval(updatePrices, 30000); updatePrices();
